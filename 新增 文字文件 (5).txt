#requires -Version 5.1
<#
.SYNOPSIS
  Extract positional text from PDFs and generate layout diagnostics, without assuming fixed column names/count.
.EXAMPLE
  .\Test-PdfLayout.ps1 -InputDirectory '.\InputPDF' -DllDirectory '.\Runtime_Test' -OutputDirectory '.\Diagnostics' -MaxPages 2
.EXAMPLE
  .\Test-PdfLayout.ps1 -SelectFiles -DllDirectory '.\Runtime_Test' -OutputDirectory '.\Diagnostics' -MaxPages 2
.EXAMPLE
  .\Test-PdfLayout.ps1 -WordsCsv '.\PdfWords.csv' -OutputDirectory '.\Diagnostics'
.NOTES
  PowerShell 5.1; uses existing PdfPig DLLs. This is a diagnostic tool, not a guaranteed table parser.
#>
[CmdletBinding(DefaultParameterSetName='Pdf')]
param(
    [Parameter(ParameterSetName='Pdf')][string]$InputDirectory = (Join-Path $PSScriptRoot 'InputPDF'),
    [Parameter(ParameterSetName='Pdf')][switch]$SelectFiles,
    [Parameter(ParameterSetName='Pdf')][string]$DllDirectory = (Join-Path $PSScriptRoot 'Runtime_Test'),
    [Parameter(Mandatory=$true,ParameterSetName='Csv')][string]$WordsCsv,
    [string]$OutputDirectory = (Join-Path $PSScriptRoot 'Diagnostics'),
    [ValidateRange(1,10000)][int]$MaxPages = 2,
    [ValidateRange(0.1,20)][double]$YTolerance = 2.5,
    [ValidateRange(0.1,30)][double]$XTolerance = 4.0
)
$ErrorActionPreference = 'Stop'
$invariant = [System.Globalization.CultureInfo]::InvariantCulture
New-Item -ItemType Directory -Force -Path $OutputDirectory | Out-Null

function Convert-Number($v) {
    return [double]::Parse(([string]$v),[System.Globalization.NumberStyles]::Float,$invariant)
}
function Write-Utf8Csv($items,$path) {
    @($items) | Export-Csv -LiteralPath $path -NoTypeInformation -Encoding UTF8
}
function Get-Lines($wordItems) {
    $lines = New-Object 'System.Collections.Generic.List[object]'
    $sorted = @($wordItems | Sort-Object @{Expression={-(Convert-Number $_.Y)}},@{Expression={Convert-Number $_.X}})
    foreach ($w in $sorted) {
        $y = Convert-Number $w.Y
        $found = $null
        foreach ($ln in $lines) {
            if ([math]::Abs($ln.Y - $y) -le $YTolerance) { $found = $ln; break }
        }
        if ($null -eq $found) {
            $found = [pscustomobject]@{Y=$y;Words=(New-Object 'System.Collections.ArrayList')}
            $lines.Add($found)
        }
        [void]$found.Words.Add($w)
    }
    return @($lines | Sort-Object Y -Descending)
}
function Analyze-Report($reportName,$words) {
    $safeName = $reportName -replace '[^a-zA-Z0-9_\-]','_'
    $reportFolder = Join-Path $OutputDirectory $safeName
    New-Item -ItemType Directory -Force -Path $reportFolder | Out-Null
    $allWords = @($words)
    if ($allWords.Count -eq 0) { Write-Warning "No words: $reportName"; return }
    Write-Utf8Csv $allWords (Join-Path $reportFolder 'Words.csv')
    $lineOut = New-Object 'System.Collections.Generic.List[object]'
    $clusterOut = New-Object 'System.Collections.Generic.List[object]'
    $footerOut = New-Object 'System.Collections.Generic.List[object]'
    foreach ($pg in ($allWords | Group-Object Page | Sort-Object {[int]$_.Name})) {
        $lines = @(Get-Lines $pg.Group)
        $lineNo=0
        foreach ($ln in $lines) {
            $lineNo++
            $ww = @($ln.Words | Sort-Object {Convert-Number $_.X})
            $txt = ($ww | ForEach-Object {$_.Text}) -join ' '
            $minX = ($ww | Measure-Object -Property X -Minimum).Minimum
            $lineOut.Add([pscustomobject]@{Page=[int]$pg.Name;Line=$lineNo;Y=$ln.Y;WordCount=$ww.Count;MinX=$minX;Text=$txt})
            if ($txt -match '(?i)\bpage\s*:\s*\d+\b') {
                $footerOut.Add([pscustomobject]@{Page=[int]$pg.Name;Y=$ln.Y;Text=$txt})
            }
        }
        # X-position clustering across words: candidates only; multi-word headings also create clusters.
        $clusters = New-Object 'System.Collections.Generic.List[object]'
        foreach ($w in @($pg.Group | Sort-Object {Convert-Number $_.X})) {
            $x=Convert-Number $w.X
            $nearest=$null; $dist=[double]::PositiveInfinity
            foreach ($c in $clusters) {
                $d=[math]::Abs($c.Center-$x)
                if ($d -le $XTolerance -and $d -lt $dist) {$nearest=$c;$dist=$d}
            }
            if ($null -eq $nearest) {
                $nearest=[pscustomobject]@{Center=$x;Count=0;Sum=0.0;Samples=(New-Object 'System.Collections.ArrayList')}
                $clusters.Add($nearest)
            }
            $nearest.Count++
            $nearest.Sum += $x
            $nearest.Center=$nearest.Sum/$nearest.Count
            if ($nearest.Samples.Count -lt 5) {[void]$nearest.Samples.Add([string]$w.Text)}
        }
        foreach ($c in @($clusters | Sort-Object Count -Descending | Select-Object -First 40)) {
            $clusterOut.Add([pscustomobject]@{Page=[int]$pg.Name;X=[math]::Round($c.Center,2);Occurrences=$c.Count;Examples=($c.Samples -join ' | ')})
        }
    }
    Write-Utf8Csv $lineOut (Join-Path $reportFolder 'VisualLines.csv')
    Write-Utf8Csv $clusterOut (Join-Path $reportFolder 'XCandidates.csv')
    Write-Utf8Csv $footerOut (Join-Path $reportFolder 'Footers.csv')
    $summary = [ordered]@{
        Report=$reportName; PagesAnalyzed=@($allWords | Select-Object -ExpandProperty Page -Unique).Count
        Words=$allWords.Count; VisualLines=$lineOut.Count
        FooterMatches=$footerOut.Count
        Notes=@('XCandidates are frequent word start positions, NOT confirmed columns.',
                'VisualLines preserves line-level text for comparing different report layouts.',
                'No row/column inference is asserted in this diagnostic version.')
    }
    $summary | ConvertTo-Json -Depth 5 | Set-Content -LiteralPath (Join-Path $reportFolder 'Summary.json') -Encoding UTF8
    Write-Host "[$reportName] Words=$($allWords.Count), Lines=$($lineOut.Count), Footers=$($footerOut.Count)" -ForegroundColor Green
}

if ($PSCmdlet.ParameterSetName -eq 'Csv') {
    if (-not (Test-Path -LiteralPath $WordsCsv -PathType Leaf)) {throw "CSV not found: $WordsCsv"}
    $raw=@(Import-Csv -LiteralPath $WordsCsv)
    if ($raw.Count -eq 0) {throw 'CSV is empty'}
    foreach ($required in @('Page','Text','X','Y')) {
        if (-not ($raw[0].PSObject.Properties.Name -contains $required)) {throw "CSV missing column: $required"}
    }
    Analyze-Report ([IO.Path]::GetFileNameWithoutExtension($WordsCsv)) $raw
    return
}

# Allow selecting PDFs from anywhere, without copying them to InputPDF.
if ($SelectFiles) {
    Add-Type -AssemblyName System.Windows.Forms
    $dialog = New-Object System.Windows.Forms.OpenFileDialog
    $dialog.Title = 'Select PDF reports (Ctrl/Shift for multiple files)'
    $dialog.Filter = 'PDF files (*.pdf)|*.pdf'
    $dialog.Multiselect = $true
    $dialog.CheckFileExists = $true
    $dialog.RestoreDirectory = $true
    try {
        if ($dialog.ShowDialog() -ne [System.Windows.Forms.DialogResult]::OK) {
            Write-Host 'No files selected; cancelled.' -ForegroundColor Yellow
            return
        }
        $selectedPdfPaths = @($dialog.FileNames)
    } finally { $dialog.Dispose() }
} else {
    if (-not (Test-Path -LiteralPath $InputDirectory -PathType Container)) {
        throw "PDF input folder not found: $InputDirectory (or use -SelectFiles)"
    }
    $selectedPdfPaths = @(Get-ChildItem -LiteralPath $InputDirectory -Filter '*.pdf' -File | Sort-Object Name | Select-Object -ExpandProperty FullName)
}
if ($selectedPdfPaths.Count -eq 0) {throw 'No PDF files selected or found.'}
if (-not (Test-Path -LiteralPath $DllDirectory -PathType Container)) {throw "DLL folder not found: $DllDirectory"}
$files = @(Get-ChildItem -LiteralPath $DllDirectory -Filter '*.dll' -File)
if ($files.Count -eq 0) {throw "No DLLs in $DllDirectory"}
# Register resolver for dependencies co-located with PdfPig (compatible with PS 5.1 / .NET Framework).
$script:PdfPigDllDirectory = (Resolve-Path -LiteralPath $DllDirectory).Path
$resolveHandler = [System.ResolveEventHandler]{
    param($sender,$args)
    $simpleName = (New-Object System.Reflection.AssemblyName($args.Name)).Name
    $candidate = Join-Path $script:PdfPigDllDirectory ($simpleName + '.dll')
    if (Test-Path -LiteralPath $candidate) {
        return [System.Reflection.Assembly]::LoadFrom($candidate)
    }
    return $null
}
[AppDomain]::CurrentDomain.add_AssemblyResolve($resolveHandler)
try {
    # Load dependencies first, then PdfPig assemblies. This avoids common PS 5.1 resolution errors.
    $loadOrder = @('System.Numerics.Vectors.dll','System.Runtime.CompilerServices.Unsafe.dll',
        'System.Buffers.dll','System.Memory.dll','Microsoft.Bcl.HashCode.dll')
    foreach ($dll in $loadOrder) {
        $path = Join-Path $script:PdfPigDllDirectory $dll
        if (Test-Path -LiteralPath $path) {[void][System.Reflection.Assembly]::LoadFrom($path)}
    }
    foreach ($f in @($files | Where-Object {$_.Name -like 'UglyToad.PdfPig*.dll'} | Sort-Object Name)) {
        [void][System.Reflection.Assembly]::LoadFrom($f.FullName)
    }
    if (-not ('UglyToad.PdfPig.PdfDocument' -as [type])) {throw 'PdfPig type not found. Check DLL directory.'}
    $pdfs=@($selectedPdfPaths | ForEach-Object {Get-Item -LiteralPath $_})
    foreach ($pdfFile in $pdfs) {
        Write-Host "Reading $($pdfFile.Name)" -ForegroundColor Cyan
        $wordList = New-Object 'System.Collections.Generic.List[object]'
        $doc=$null
        try {
            $doc=[UglyToad.PdfPig.PdfDocument]::Open($pdfFile.FullName)
            $count=0
            foreach ($page in $doc.GetPages()) {
                $count++
                if ($count -gt $MaxPages) {break}
                foreach ($w in $page.GetWords()) {
                    $wordList.Add([pscustomobject]@{
                        Page=$page.Number;Text=$w.Text
                        X=[math]::Round($w.BoundingBox.Left,2)
                        Y=[math]::Round($w.BoundingBox.Bottom,2)
                        Width=[math]::Round(($w.BoundingBox.Right-$w.BoundingBox.Left),2)
                        Height=[math]::Round(($w.BoundingBox.Top-$w.BoundingBox.Bottom),2)
                    })
                }
            }
            Analyze-Report $pdfFile.BaseName $wordList
        } catch {
            Write-Warning "Failed $($pdfFile.Name): $($_.Exception.Message)"
            [pscustomobject]@{File=$pdfFile.Name;Error=$_.Exception.ToString()} |
                ConvertTo-Json -Depth 3 | Set-Content -LiteralPath (Join-Path $OutputDirectory ($pdfFile.BaseName + '_Error.json')) -Encoding UTF8
        } finally {if ($null -ne $doc) {$doc.Dispose()}}
    }
} finally {
    [AppDomain]::CurrentDomain.remove_AssemblyResolve($resolveHandler)
}
