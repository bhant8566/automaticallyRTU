$ErrorActionPreference = "Stop"

[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12

$root = "$env:USERPROFILE\Desktop\PdfPigTest"
$depRoot = Join-Path $root "Dependencies"
$runtime = Join-Path $root "Runtime"

$name = "system.runtime.compilerservices.unsafe"
$version = "6.1.0"

$url = "https://api.nuget.org/v3-flatcontainer/$name/$version/$name.$version.nupkg"

$packagePath = Join-Path $depRoot "$name.$version.nupkg"
$extractPath = Join-Path $depRoot "$name.$version"

# 下載套件
Invoke-WebRequest -Uri $url `
    -OutFile $packagePath `
    -UseBasicParsing

# 解壓縮
Add-Type -AssemblyName System.IO.Compression.FileSystem

if (-not (Test-Path $extractPath)) {
    [System.IO.Compression.ZipFile]::ExtractToDirectory(
        $packagePath,
        $extractPath
    )
}

# 尋找適合 .NET Framework 的 DLL
$dllPath = Join-Path $extractPath `
    "lib\net462\System.Runtime.CompilerServices.Unsafe.dll"

if (-not (Test-Path $dllPath)) {
    Write-Host "net462 DLL not found. Available files:" -ForegroundColor Yellow

    Get-ChildItem "$extractPath\lib" -Recurse -Filter "*.dll" |
        Select-Object FullName

    throw "Please verify the correct target framework."
}

# 檢查 Assembly Version
$assembly = [System.Reflection.AssemblyName]::GetAssemblyName($dllPath)

Write-Host "Assembly: $($assembly.Name)"
Write-Host "Version: $($assembly.Version)"

# 複製到 Runtime
Copy-Item $dllPath -Destination $runtime -Force

Write-Host "DLL copied successfully." -ForegroundColor Green

$runtime = "$env:USERPROFILE\Desktop\PdfPigTest\Runtime"

$dependencies = @(
    "System.Runtime.CompilerServices.Unsafe.dll",
    "System.Memory.dll",
    "Microsoft.Bcl.HashCode.dll"
)

foreach ($dll in $dependencies) {
    $path = Join-Path $runtime $dll

    $assembly = [System.Reflection.Assembly]::LoadFrom($path)

    Write-Host "Loaded: $($assembly.GetName().Name)" -ForegroundColor Green
}

# 載入 PdfPig
[System.Reflection.Assembly]::LoadFrom(
    (Join-Path $runtime "UglyToad.PdfPig.dll")
) | Out-Null

Write-Host "PdfPig loaded." -ForegroundColor Green

$pdfPath = "C:\Temp\Report.pdf"

$pdf = [UglyToad.PdfPig.PdfDocument]::Open($pdfPath)

try {
    $page = $pdf.GetPage(1)

    $page.GetWords() |
        Select-Object -First 20 |
        ForEach-Object {
            [PSCustomObject]@{
                Text = $_.Text
                X    = $_.BoundingBox.Left
                Y    = $_.BoundingBox.Bottom
            }
        } |
        Format-Table -AutoSize
}
finally {
    $pdf.Dispose()
}
