function Get-FrameworkRank {
    param([string]$Name)

    if ([string]::IsNullOrWhiteSpace($Name)) {
        return 0
    }

    $n = $Name.Trim().ToLowerInvariant()

    # NuGet Framework names:
    # .NETFramework4.7.1
    # .NETFramework,Version=v4.7.1
    # net471
    # net462

    if ($n -match '^\.netframework(?:,version=v)?(\d+)\.(\d+)(?:\.(\d+))?$') {

        $major = [int]$matches[1]
        $minor = [int]$matches[2]
        $patch = 0

        if ($matches[3]) {
            $patch = [int]$matches[3]
        }

        $rank = $major * 100 + $minor * 10 + $patch

        if ($rank -le 480) {
            return $rank
        }

        return -1
    }

    if ($n -match '^net(4)(\d)(\d?)$') {

        $major = [int]$matches[1]
        $minor = [int]$matches[2]
        $patch = 0

        if ($matches[3]) {
            $patch = [int]$matches[3]
        }

        $rank = $major * 100 + $minor * 10 + $patch

        if ($rank -le 480) {
            return $rank
        }

        return -1
    }

    # Reject .NET 6/8/9 and unsupported frameworks
    return -1
}
