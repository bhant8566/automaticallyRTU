$ErrorActionPreference = 'Stop'

[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12

$root = Join-Path $env:USERPROFILE 'Desktop\PdfPigTest'
$dependencies = Join-Path $root 'Dependencies'
$runtime = Join-Path $root 'Runtime_Test'

$name = 'system.numerics.vectors'
$version = '4.6.0'

$packagePath = Join-Path $dependencies "$name.$version.nupkg"
$extractPath = Join-Path $dependencies "$name.$version"

$url = "https://api.nuget.org/v3-flatcontainer/$name/$version/$name.$version.nupkg"

New-Item -ItemType Directory -Force -Path $dependencies, $runtime | Out-Null

# Download only if not already present
if (-not (Test-Path $packagePath)) {
    Invoke-WebRequest `
        -Uri $url `
        -OutFile $packagePath `
        -UseBasicParsing
}

# Extract NuGet package
Add-Type -AssemblyName System.IO.Compression.FileSystem

if (-not (Test-Path $extractPath)) {
    [System.IO.Compression.ZipFile]::ExtractToDirectory(
        $packagePath,
        $extractPath
    )
}

# Find DLL with required Assembly Version
$expectedVersion = '4.1.5.0'

$candidates = @(
    Get-ChildItem (Join-Path $extractPath 'lib') `
        -Recurse `
        -Filter 'System.Numerics.Vectors.dll' `
        -File
)

$matched = $null

foreach ($file in $candidates) {

    $assembly = [Reflection.AssemblyName]::GetAssemblyName($file.FullName)

    Write-Host "$($assembly.Version) : $($file.FullName)"

    if ($assembly.Version.ToString() -eq $expectedVersion) {

        # Prefer .NET Framework assembly
        if ($file.FullName -match '\\lib\\net4\d+\\') {
            $matched = $file
            break
        }

        if (-not $matched) {
            $matched = $file
        }
    }
}

if (-not $matched) {
    throw "Required Assembly Version $expectedVersion was not found."
}

Copy-Item $matched.FullName -Destination $runtime -Force

Write-Host ''
Write-Host "Installed: System.Numerics.Vectors $expectedVersion" `
    -ForegroundColor Green
Write-Host "Source: $($matched.FullName)"
Write-Host "Destination: $runtime"
