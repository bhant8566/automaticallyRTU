$ErrorActionPreference = "Stop"

$runtime = "$env:USERPROFILE\Desktop\PdfPigTest\Runtime"

try {
    # 先載入目前已知的相依 DLL
    $dependencies = @(
        "System.Memory.dll",
        "Microsoft.Bcl.HashCode.dll"
    )

    foreach ($dll in $dependencies) {
        $path = Join-Path $runtime $dll

        $assembly = [System.Reflection.Assembly]::LoadFrom($path)

        Write-Host "Loaded: $($assembly.GetName().Name)" `
            -ForegroundColor Green
    }

    # 載入 PdfPig 主組件
    $pdfPigPath = Join-Path $runtime "UglyToad.PdfPig.dll"

    $pdfAssembly = [System.Reflection.Assembly]::LoadFrom($pdfPigPath)

    Write-Host "Loaded: $($pdfAssembly.GetName().Name)" `
        -ForegroundColor Green

    # 驗證主類別
    $pdfType = $pdfAssembly.GetType(
        "UglyToad.PdfPig.PdfDocument",
        $true
    )

    Write-Host "PdfDocument type found: $($pdfType.FullName)" `
        -ForegroundColor Green
}
catch {
    Write-Host "Loading failed:" -ForegroundColor Red
    Write-Host $_.Exception.ToString()
}

$pdfPath = "C:\Temp\Report.pdf"

$pdf = [UglyToad.PdfPig.PdfDocument]::Open($pdfPath)

try {
    $page = $pdf.GetPage(1)

    $page.GetWords() |
        Select-Object -First 20 |
        ForEach-Object {
            [PSCustomObject]@{
                Text = $_.Text
                X    = $_.BoundingBox.Left
                Y    = $_.BoundingBox.Bottom
            }
        } |
        Format-Table -AutoSize
}
finally {
    $pdf.Dispose()
}
